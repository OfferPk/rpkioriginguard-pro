/\*\*  
 \* RPKIOriginGuard Pro \- Multi-Router Zero-Trust BGP/RPKI Synthesizer  
 \* Generates:  
 \* 1\. BIRD 2 RPKI protocol & filter configuration  
 \* 2\. FRRouting (FRR) RPKI & route-map policy  
 \* 3\. Cisco IOS-XR BGP ROV policy  
 \* 4\. Juniper Junos validation group & import policy  
 \*/

(function (root, factory) {  
  if (typeof define \=== 'function' && define.amd) {  
    define(\[\], factory);  
  } else if (typeof module \=== 'object' && module.exports) {  
    module.exports \= factory();  
  } else {  
    root.RouterSynthesizer \= factory();  
  }  
})(typeof self \!== 'undefined' ? self : this, function () {  
  'use strict';

  function generateBird2Config(config) {  
    const asn \= config.asn || 64512;  
    const rtrServers \= (config.routerConfig && config.routerConfig.rtrValidators) || \['10.0.1.50:3323'\];  
    const rtr1 \= rtrServers\[0\] || '127.0.0.1:3323';  
    const \[ip1, port1\] \= rtr1.split(':');

    return \`\# \==============================================================================  
\# BIRD 2 Hardened RPKI Route Origin Validation (ROV) Configuration  
\# Generated by RPKIOriginGuard Pro (RFC 6811 & RFC 8210\)  
\# \==============================================================================

\# Define ROA tables for IPv4 and IPv6  
roa4 table r4;  
roa6 table r6;

\# RPKI-to-Router (RTR) Protocol Instance  
protocol rpki rpki\_primary {  
    roa4 { table r4; };  
    roa6 { table r6; };  
    remote "\${ip1}" port \${port1 || 3323};  
    retry 30;  
    refresh 600;  
    expire 7200;  
}

\# Strict Zero-Trust ROV Ingress Filter  
function check\_rpki\_rov() {  
    \# Check IPv4 ROA table  
    if net.type \= NET\_IP4 then {  
        if roa\_check(r4, net, bgp\_path.last) \= ROA\_INVALID then {  
            print "\[RPKI REJECT\] Dropping invalid IPv4 route: ", net, " from AS", bgp\_path.last;  
            reject;  
        }  
    }  
    \# Check IPv6 ROA table  
    if net.type \= NET\_IP6 then {  
        if roa\_check(r6, net, bgp\_path.last) \= ROA\_INVALID then {  
            print "\[RPKI REJECT\] Dropping invalid IPv6 route: ", net, " from AS", bgp\_path.last;  
            reject;  
        }  
    }  
}

\# BGP Template applying ROV  
template bgp ebgp\_transit {  
    local as \${asn};  
    ipv4 {  
        import filter {  
            check\_rpki\_rov();  
            accept;  
        };  
        export filter {  
            \# Only announce authorized local prefixes  
            accept;  
        };  
    };  
}  
\`;  
  }

  function generateFrrConfig(config) {  
    const asn \= config.asn || 64512;  
    const rtrServers \= (config.routerConfig && config.routerConfig.rtrValidators) || \['10.0.1.50:3323'\];  
    const rtr1 \= rtrServers\[0\] || '127.0.0.1:3323';  
    const \[ip1, port1\] \= rtr1.split(':');

    return \`\# \==============================================================================  
\# FRRouting (FRR) Hardened RPKI ROV & Route-Map Configuration  
\# Generated by RPKIOriginGuard Pro  
\# \==============================================================================

\! RPKI RTR Cache Client Configuration  
rpki  
  rpki cache \${ip1} \${port1 || 3323} preference 1  
  rpki initial-synchronisation-timeout 30  
\!

\! Route-Map enforcing Drop-Invalid and Preference Gating  
route-map EBGP-IN deny 10  
  match rpki invalid  
\!  
route-map EBGP-IN permit 20  
  match rpki valid  
  set local-preference 200  
\!  
route-map EBGP-IN permit 30  
  match rpki notfound  
  set local-preference 100  
\!

\! Router BGP Configuration  
router bgp \${asn}  
  bgp router-id 198.51.100.1  
  neighbor TRANSIT-PEERS peer-group  
  neighbor TRANSIT-PEERS route-map EBGP-IN in  
\!  
\`;  
  }

  function generateCiscoConfig(config) {  
    const asn \= config.asn || 64512;  
    const rtrServers \= (config.routerConfig && config.routerConfig.rtrValidators) || \['10.0.1.50:3323'\];  
    const rtr1 \= rtrServers\[0\] || '10.0.1.50:3323';  
    const \[ip1, port1\] \= rtr1.split(':');

    return \`\! \==============================================================================  
\! Cisco IOS-XR Zero-Trust RPKI Configuration & RPL Route Policy  
\! Generated by RPKIOriginGuard Pro  
\! \==============================================================================

router bgp \${asn}  
  bgp origin-as validation enable  
  rpki server \${ip1}  
    transport tcp port \${port1 || 3323}  
    refresh-time 600  
  \!  
\!

route-policy RPKI-INGRESS-FILTER  
  \# Immediately discard RPKI Invalid routes  
  if roa-state is invalid then  
    drop  
  elseif roa-state is valid then  
    set local-preference 200  
    pass  
  else  
    \# NotFound / Unknown  
    set local-preference 100  
    pass  
  endif  
end-policy  
\`;  
  }

  function generateJunosConfig(config) {  
    const asn \= config.asn || 64512;  
    const rtrServers \= (config.routerConfig && config.routerConfig.rtrValidators) || \['10.0.1.50:3323'\];  
    const rtr1 \= rtrServers\[0\] || '10.0.1.50:3323';  
    const \[ip1, port1\] \= rtr1.split(':');

    return \`\# \==============================================================================  
\# Juniper Junos RPKI Validation Group & Import Policy  
\# Generated by RPKIOriginGuard Pro  
\# \==============================================================================

routing-options {  
    validation {  
        notification-rib (inet.0, inet6.0);  
        group rpki-validators {  
            session \${ip1} {  
                port \${port1 || 3323};  
                local-address 10.0.1.1;  
                refresh-time 600;  
            }  
        }  
    }  
}

policy-options {  
    policy-statement RPKI-IMPORT-POLICY {  
        term drop-invalid {  
            from {  
                validation-database invalid;  
            }  
            then {  
                reject;  
            }  
        }  
        term accept-valid {  
            from {  
                validation-database valid;  
            }  
            then {  
                local-preference 200;  
                accept;  
            }  
        }  
        term default-accept {  
            then {  
                local-preference 100;  
                accept;  
            }  
        }  
    }  
}  
\`;  
  }

  return {  
    generateBird2Config,  
    generateFrrConfig,  
    generateCiscoConfig,  
    generateJunosConfig  
  };  
});
